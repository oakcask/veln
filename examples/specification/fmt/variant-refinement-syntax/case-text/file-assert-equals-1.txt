fn transition(state: Alias::Ready | protocol::State::Closed | Alias::Ready, container: Alias::Container<Int>, callback: fn(State::Open | State::Closed) -> State::Open, rest: ...Result<List<Int>, Error>::Err) -> Result<List<Int>, Error>::Ok
	let nested: { states : List<State::Open | State::Closed> } = state
	sink<State::Open | State::Closed>(nested)
	sink<State::Ready | State::Closed, Result<State::Ready | State::Closed, Error>::Ok>(nested)
	sink<List<State::Ready>>(nested)
end
